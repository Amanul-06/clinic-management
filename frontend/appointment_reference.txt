import { lazy, Suspense } from "react";
import { Navigate, Route, Routes, Outlet } from "react-router-dom";

import { useAuth } from "./context/useAuth";
import DashboardLayout from "./layouts/DashboardLayout/DashboardLayout";

const Login = lazy(() => import("./pages/Login/Login"));
const Dashboard = lazy(() => import("./pages/Dashboard/Dashboard"));
const Patients = lazy(() => import("./pages/Patients/Patients"));
const Doctors = lazy(() => import("./pages/Doctors/Doctors"));
const Settings = lazy(() => import("./pages/Settings/Settings"));

function ProtectedRoute() {
  const { user } = useAuth();

  return user ? <DashboardLayout /> : <Navigate to="/login" replace />;
}

function AdminRoute() {
  const { user } = useAuth();

  if (!user) return <Navigate to="/login" replace />;

  return user.role === "superadmin" ? (
    <Outlet />
  ) : (
    <Navigate to="/dashboard" replace />
  );
}

export default function App() {
  return (
    <Suspense fallback={<div className="route-loading">Loading...</div>}>
      <Routes>
        <Route path="/login" element={<Login />} />

        <Route element={<ProtectedRoute />}>
          <Route path="/dashboard" element={<Dashboard />} />
          <Route path="/patients" element={<Patients />} />
          <Route path="/doctors" element={<Doctors />} />

          <Route element={<AdminRoute />}>
            <Route path="/settings" element={<Settings />} />
          </Route>
        </Route>

        <Route path="/" element={<Navigate to="/dashboard" replace />} />
        <Route path="*" element={<Navigate to="/dashboard" replace />} />
      </Routes>
    </Suspense>
  );
}
import { useEffect, useMemo, useState } from "react";
import {
  Search,
  Plus,
  Stethoscope,
  Pencil,
  Trash2,
  X,
  CalendarDays,
  UserRound,
  Clock3,
  PlusCircle,
  Trash,
} from "lucide-react";

import { demoStore } from "../../data/demoStore";
import "./Doctors.css";

const weekdays = [
  "Monday",
  "Tuesday",
  "Wednesday",
  "Thursday",
  "Friday",
  "Saturday",
  "Sunday",
];

const emptyDoctor = {
  name: "",
  specialization: "",
  qualification: "",
  experience: "",
  phone: "",
  consultationFee: "",
  status: "Active",
};

const createShift = () => ({
  startTime: "09:00",
  endTime: "13:00",
  slotDuration: 30,
});

const createWeeklySchedule = () =>
  Object.fromEntries(
    weekdays.map((day) => [
      day,
      {
        enabled: false,
        shifts: [],
      },
    ]),
  );

function normalizeSchedule(existing) {
  const weekly = createWeeklySchedule();

  if (!existing) return weekly;

  // Support schedules created using the previous schedule format.
  if (existing.weekly) {
    weekdays.forEach((day) => {
      const savedDay = existing.weekly[day];

      if (savedDay) {
        weekly[day] = {
          enabled: Boolean(savedDay.enabled),
          shifts: Array.isArray(savedDay.shifts)
            ? savedDay.shifts.map((shift) => ({
                startTime: shift.startTime || "09:00",
                endTime: shift.endTime || "13:00",
                slotDuration: Number(shift.slotDuration) || 30,
              }))
            : [],
        };
      }
    });

    return weekly;
  }

  if (Array.isArray(existing.days)) {
    existing.days.forEach((day) => {
      if (weekly[day]) {
        weekly[day] = {
          enabled: true,
          shifts: [
            {
              startTime: existing.startTime || "09:00",
              endTime: existing.endTime || "17:00",
              slotDuration: Number(existing.slotDuration) || 30,
            },
          ],
        };
      }
    });
  }

  return weekly;
}

function formatTime(time) {
  if (!time) return "";

  const [hours, minutes] = time.split(":").map(Number);
  const suffix = hours >= 12 ? "PM" : "AM";
  const displayHour = hours % 12 || 12;

  return `${displayHour}:${String(minutes).padStart(2, "0")} ${suffix}`;
}

function getScheduleSummary(schedule) {
  if (!schedule) return "Not configured";

  const weekly = normalizeSchedule(schedule);
  const workingDays = weekdays.filter(
    (day) => weekly[day].enabled && weekly[day].shifts.length,
  );

  if (!workingDays.length) return "Not configured";

  return `${workingDays.length} working days`;
}

export default function Doctors() {
  const [doctors, setDoctors] = useState([]);
  const [schedules, setSchedules] = useState([]);
  const [search, setSearch] = useState("");

  const [showModal, setShowModal] = useState(false);
  const [showSchedule, setShowSchedule] = useState(false);

  const [editingId, setEditingId] = useState(null);
  const [selectedDoctor, setSelectedDoctor] = useState(null);

  const [form, setForm] = useState({ ...emptyDoctor });
  const [weeklySchedule, setWeeklySchedule] = useState(createWeeklySchedule);

  const loadData = () => {
    setDoctors(demoStore.getAll("doctors"));
    setSchedules(demoStore.getAll("schedules"));
  };

  useEffect(() => {
    loadData();
    return demoStore.subscribe(loadData);
  }, []);

  const getSchedule = (doctorId) =>
    schedules.find((item) => item.doctorId === doctorId);

  const filteredDoctors = useMemo(() => {
    const query = search.toLowerCase().trim();

    if (!query) return doctors;

    return doctors.filter((doctor) =>
      [doctor.id, doctor.name, doctor.specialization, doctor.phone].some(
        (value) =>
          String(value || "")
            .toLowerCase()
            .includes(query),
      ),
    );
  }, [doctors, search]);

  const activeCount = doctors.filter(
    (doctor) => doctor.status === "Active",
  ).length;

  const scheduledCount = doctors.filter((doctor) => {
    const saved = getSchedule(doctor.id);
    return saved && getScheduleSummary(saved) !== "Not configured";
  }).length;

  const openAddModal = () => {
    setEditingId(null);
    setForm({ ...emptyDoctor });
    setShowModal(true);
  };

  const openEditModal = (doctor) => {
    setEditingId(doctor.id);

    setForm({
      name: doctor.name || "",
      specialization: doctor.specialization || "",
      qualification: doctor.qualification || "",
      experience: doctor.experience ?? "",
      phone: doctor.phone || "",
      consultationFee: doctor.consultationFee ?? "",
      status: doctor.status || "Active",
    });

    setShowModal(true);
  };

  const handleChange = (event) => {
    const { name, value } = event.target;

    setForm((previous) => ({
      ...previous,
      [name]: value,
    }));
  };

  const handleSubmit = (event) => {
    event.preventDefault();

    if (!form.name.trim() || !form.specialization.trim()) {
      alert("Doctor name and specialization are required.");
      return;
    }

    if (form.phone && !/^\d{10}$/.test(form.phone)) {
      alert("Enter a valid 10-digit phone number.");
      return;
    }

    const payload = {
      ...form,
      name: form.name.trim(),
      specialization: form.specialization.trim(),
      experience: Number(form.experience) || 0,
      consultationFee: Number(form.consultationFee) || 0,
    };

    const result = editingId
      ? demoStore.update("doctors", editingId, payload)
      : demoStore.add("doctors", payload, "DOC");

    if (!result) {
      alert("Unable to save doctor.");
      return;
    }

    setShowModal(false);
    setForm({ ...emptyDoctor });
    loadData();
  };

  const handleDelete = (doctor) => {
    const hasAppointments = demoStore
      .getAll("appointments")
      .some(
        (appointment) =>
          appointment.doctorId === doctor.id &&
          appointment.status !== "Cancelled",
      );

    if (hasAppointments) {
      alert(
        "This doctor has appointments. Please mark the doctor inactive instead.",
      );
      return;
    }

    if (!window.confirm(`Delete ${doctor.name}?`)) return;

    demoStore.delete("doctors", doctor.id);

    const savedSchedule = getSchedule(doctor.id);

    if (savedSchedule) {
      demoStore.delete("schedules", savedSchedule.id);
    }

    loadData();
  };

  const openSchedule = (doctor) => {
    setSelectedDoctor(doctor);

    const existing = getSchedule(doctor.id);
    setWeeklySchedule(normalizeSchedule(existing));

    setShowSchedule(true);
  };

  const toggleDay = (day) => {
    setWeeklySchedule((previous) => ({
      ...previous,
      [day]: {
        ...previous[day],
        enabled: !previous[day].enabled,
        shifts: !previous[day].enabled
          ? previous[day].shifts.length
            ? previous[day].shifts
            : [createShift()]
          : previous[day].shifts,
      },
    }));
  };

  const addShift = (day) => {
    setWeeklySchedule((previous) => ({
      ...previous,
      [day]: {
        ...previous[day],
        enabled: true,
        shifts: [...previous[day].shifts, createShift()],
      },
    }));
  };

  const removeShift = (day, shiftIndex) => {
    setWeeklySchedule((previous) => {
      const shifts = previous[day].shifts.filter(
        (_, index) => index !== shiftIndex,
      );

      return {
        ...previous,
        [day]: {
          ...previous[day],
          shifts,
          enabled: shifts.length > 0,
        },
      };
    });
  };

  const updateShift = (day, shiftIndex, field, value) => {
    setWeeklySchedule((previous) => ({
      ...previous,
      [day]: {
        ...previous[day],
        shifts: previous[day].shifts.map((shift, index) =>
          index === shiftIndex
            ? {
                ...shift,
                [field]: field === "slotDuration" ? Number(value) : value,
              }
            : shift,
        ),
      },
    }));
  };

  const validateWeeklySchedule = () => {
    let hasWorkingDay = false;

    for (const day of weekdays) {
      const daySchedule = weeklySchedule[day];

      if (!daySchedule.enabled) continue;

      if (!daySchedule.shifts.length) {
        alert(`${day}: Add at least one shift or mark this day as off.`);
        return false;
      }

      hasWorkingDay = true;

      const sortedShifts = [...daySchedule.shifts].sort((a, b) =>
        a.startTime.localeCompare(b.startTime),
      );

      for (let index = 0; index < sortedShifts.length; index += 1) {
        const shift = sortedShifts[index];

        if (!shift.startTime || !shift.endTime) {
          alert(`${day}: Please enter both start and end times.`);
          return false;
        }

        if (shift.startTime >= shift.endTime) {
          alert(`${day}: End time must be later than start time.`);
          return false;
        }

        if (index > 0 && sortedShifts[index - 1].endTime > shift.startTime) {
          alert(`${day}: Your shifts overlap. Please correct the timings.`);
          return false;
        }

        if (!shift.slotDuration || shift.slotDuration < 5) {
          alert(`${day}: Appointment duration must be at least 5 minutes.`);
          return false;
        }
      }
    }

    if (!hasWorkingDay) {
      alert("Configure at least one working day.");
      return false;
    }

    return true;
  };

  const saveSchedule = (event) => {
    event.preventDefault();

    if (!validateWeeklySchedule()) return;

    const existing = getSchedule(selectedDoctor.id);

    const payload = {
      doctorId: selectedDoctor.id,
      weekly: weeklySchedule,
    };

    const result = existing
      ? demoStore.update("schedules", existing.id, payload)
      : demoStore.add("schedules", payload, "SCH");

    if (!result) {
      alert("Unable to save schedule.");
      return;
    }

    setShowSchedule(false);
    loadData();
  };

  return (
    <div className="doctors-page">
      <div className="doctors-header">
        <div>
          <span className="doctors-eyebrow">CLINIC MANAGEMENT</span>
          <h1>Doctors</h1>
          <p>Manage doctors, consultation fees and weekly availability.</p>
        </div>

        <button className="doctor-primary-btn" onClick={openAddModal}>
          <Plus size={18} />
          Add Doctor
        </button>
      </div>

      <div className="doctors-stats">
        <div className="doctor-stat-card">
          <div className="doctor-stat-icon blue">
            <Stethoscope size={21} />
          </div>
          <div>
            <span>Total Doctors</span>
            <strong>{doctors.length}</strong>
          </div>
        </div>

        <div className="doctor-stat-card">
          <div className="doctor-stat-icon green">
            <UserRound size={21} />
          </div>
          <div>
            <span>Active Doctors</span>
            <strong>{activeCount}</strong>
          </div>
        </div>

        <div className="doctor-stat-card">
          <div className="doctor-stat-icon purple">
            <CalendarDays size={21} />
          </div>
          <div>
            <span>With Schedules</span>
            <strong>{scheduledCount}</strong>
          </div>
        </div>
      </div>

      <section className="doctors-content">
        <div className="doctors-toolbar">
          <div>
            <h2>All Doctors</h2>
            <p>{filteredDoctors.length} doctors found</p>
          </div>

          <div className="doctor-search">
            <Search size={18} />
            <input
              placeholder="Search doctors..."
              value={search}
              onChange={(event) => setSearch(event.target.value)}
            />
          </div>
        </div>

        <div className="doctors-table-wrapper">
          <table className="doctors-table">
            <thead>
              <tr>
                <th>Doctor</th>
                <th>Specialization</th>
                <th>Experience</th>
                <th>Consultation Fee</th>
                <th>Weekly Schedule</th>
                <th>Status</th>
                <th>Actions</th>
              </tr>
            </thead>

            <tbody>
              {filteredDoctors.map((doctor) => {
                const savedSchedule = getSchedule(doctor.id);

                return (
                  <tr key={doctor.id}>
                    <td>
                      <div className="doctor-name-cell">
                        <div className="doctor-avatar">
                          {doctor.name
                            ?.replace("Dr. ", "")
                            .charAt(0)
                            .toUpperCase()}
                        </div>

                        <div>
                          <strong>{doctor.name}</strong>
                          <span>{doctor.id}</span>
                        </div>
                      </div>
                    </td>

                    <td>
                      <div className="doctor-specialization">
                        {doctor.specialization}
                        <span>{doctor.qualification}</span>
                      </div>
                    </td>

                    <td>{doctor.experience} years</td>

                    <td>
                      <span className="doctor-fee">
                        ₹
                        {Number(doctor.consultationFee || 0).toLocaleString(
                          "en-IN",
                        )}
                      </span>
                    </td>

                    <td>
                      <div className="schedule-cell">
                        <span>{getScheduleSummary(savedSchedule)}</span>
                        <button
                          className="doctor-schedule-btn"
                          onClick={() => openSchedule(doctor)}
                        >
                          <CalendarDays size={14} />
                          {savedSchedule ? "Manage" : "Set schedule"}
                        </button>
                      </div>
                    </td>

                    <td>
                      <span
                        className={`doctor-status ${
                          doctor.status === "Active" ? "active" : "inactive"
                        }`}
                      >
                        {doctor.status}
                      </span>
                    </td>

                    <td>
                      <div className="doctor-actions">
                        <button
                          title="Edit doctor"
                          onClick={() => openEditModal(doctor)}
                        >
                          <Pencil size={16} />
                        </button>

                        <button
                          title="Delete doctor"
                          className="delete"
                          onClick={() => handleDelete(doctor)}
                        >
                          <Trash2 size={16} />
                        </button>
                      </div>
                    </td>
                  </tr>
                );
              })}

              {filteredDoctors.length === 0 && (
                <tr>
                  <td colSpan="7">
                    <div className="doctors-empty">
                      <Stethoscope size={32} />
                      <strong>No doctors found</strong>
                      <span>Try another search or add a doctor.</span>
                    </div>
                  </td>
                </tr>
              )}
            </tbody>
          </table>
        </div>
      </section>

      {showModal && (
        <div
          className="doctor-modal-overlay"
          onMouseDown={(event) => {
            if (event.target === event.currentTarget) {
              setShowModal(false);
            }
          }}
        >
          <div className="doctor-modal">
            <div className="doctor-modal-header">
              <div>
                <span className="doctors-eyebrow">DOCTOR MANAGEMENT</span>
                <h2>{editingId ? "Edit Doctor" : "Add Doctor"}</h2>
              </div>

              <button
                type="button"
                className="doctor-modal-close"
                onClick={() => setShowModal(false)}
              >
                <X size={20} />
              </button>
            </div>

            <form onSubmit={handleSubmit}>
              <div className="doctor-form-grid">
                <label>
                  Doctor Name *
                  <input
                    name="name"
                    value={form.name}
                    onChange={handleChange}
                    placeholder="Dr. Full Name"
                    required
                  />
                </label>

                <label>
                  Specialization *
                  <input
                    name="specialization"
                    value={form.specialization}
                    onChange={handleChange}
                    placeholder="e.g. Cardiology"
                    required
                  />
                </label>

                <label>
                  Qualification
                  <input
                    name="qualification"
                    value={form.qualification}
                    onChange={handleChange}
                    placeholder="e.g. MBBS, MD"
                  />
                </label>

                <label>
                  Experience (years)
                  <input
                    type="number"
                    min="0"
                    name="experience"
                    value={form.experience}
                    onChange={handleChange}
                    placeholder="e.g. 10"
                  />
                </label>

                <label>
                  Phone Number
                  <input
                    name="phone"
                    inputMode="numeric"
                    maxLength="10"
                    value={form.phone}
                    onChange={handleChange}
                    placeholder="10-digit phone number"
                  />
                </label>

                <label>
                  Consultation Fee (₹)
                  <input
                    type="number"
                    min="0"
                    name="consultationFee"
                    value={form.consultationFee}
                    onChange={handleChange}
                    placeholder="e.g. 500"
                  />
                </label>

                <label>
                  Status
                  <select
                    name="status"
                    value={form.status}
                    onChange={handleChange}
                  >
                    <option value="Active">Active</option>
                    <option value="Inactive">Inactive</option>
                  </select>
                </label>
              </div>

              <div className="doctor-form-footer">
                <button
                  type="button"
                  className="doctor-cancel-btn"
                  onClick={() => setShowModal(false)}
                >
                  Cancel
                </button>

                <button type="submit" className="doctor-primary-btn">
                  {editingId ? "Save Changes" : "Add Doctor"}
                </button>
              </div>
            </form>
          </div>
        </div>
      )}

      {showSchedule && selectedDoctor && (
        <div
          className="doctor-modal-overlay schedule-overlay"
          onMouseDown={(event) => {
            if (event.target === event.currentTarget) {
              setShowSchedule(false);
            }
          }}
        >
          <div className="doctor-modal schedule-modal">
            <div className="doctor-modal-header">
              <div>
                <span className="doctors-eyebrow">WEEKLY AVAILABILITY</span>
                <h2>Doctor Schedule</h2>
                <p>{selectedDoctor.name}</p>
              </div>

              <button
                type="button"
                className="doctor-modal-close"
                onClick={() => setShowSchedule(false)}
              >
                <X size={20} />
              </button>
            </div>

            <form onSubmit={saveSchedule}>
              <div className="schedule-intro">
                <CalendarDays size={20} />
                <p>
                  Set individual working hours for each day. You can add
                  multiple shifts when a doctor takes a break between
                  consultations.
                </p>
              </div>

              <div className="weekly-schedule-list">
                {weekdays.map((day) => {
                  const daySchedule = weeklySchedule[day];

                  return (
                    <section className="schedule-day-card" key={day}>
                      <div className="schedule-day-header">
                        <div className="schedule-day-name">
                          <strong>{day}</strong>
                          <span>
                            {daySchedule.enabled
                              ? `${daySchedule.shifts.length} shift(s)`
                              : "Day off"}
                          </span>
                        </div>

                        <label className="schedule-toggle">
                          <input
                            type="checkbox"
                            checked={daySchedule.enabled}
                            onChange={() => toggleDay(day)}
                          />
                          <span className="schedule-toggle-track" />
                          <span className="schedule-toggle-label">
                            {daySchedule.enabled ? "Working" : "Off"}
                          </span>
                        </label>
                      </div>

                      {daySchedule.enabled && (
                        <div className="schedule-shifts">
                          {daySchedule.shifts.map((shift, index) => (
                            <div
                              className="schedule-shift"
                              key={`${day}-${index}`}
                            >
                              <div className="schedule-shift-heading">
                                <span>
                                  <Clock3 size={15} />
                                  Shift {index + 1}
                                </span>

                                <button
                                  type="button"
                                  className="remove-shift-btn"
                                  onClick={() => removeShift(day, index)}
                                  title="Remove shift"
                                >
                                  <Trash size={15} />
                                  Remove
                                </button>
                              </div>

                              <div className="schedule-shift-fields">
                                <label>
                                  Start time
                                  <input
                                    type="time"
                                    value={shift.startTime}
                                    onChange={(event) =>
                                      updateShift(
                                        day,
                                        index,
                                        "startTime",
                                        event.target.value,
                                      )
                                    }
                                    required
                                  />
                                </label>

                                <label>
                                  End time
                                  <input
                                    type="time"
                                    value={shift.endTime}
                                    onChange={(event) =>
                                      updateShift(
                                        day,
                                        index,
                                        "endTime",
                                        event.target.value,
                                      )
                                    }
                                    required
                                  />
                                </label>

                                <label>
                                  Appointment duration
                                  <select
                                    value={shift.slotDuration}
                                    onChange={(event) =>
                                      updateShift(
                                        day,
                                        index,
                                        "slotDuration",
                                        event.target.value,
                                      )
                                    }
                                  >
                                    <option value={15}>15 minutes</option>
                                    <option value={20}>20 minutes</option>
                                    <option value={30}>30 minutes</option>
                                    <option value={45}>45 minutes</option>
                                    <option value={60}>60 minutes</option>
                                  </select>
                                </label>
                              </div>

                              <div className="shift-time-preview">
                                {formatTime(shift.startTime)} –{" "}
                                {formatTime(shift.endTime)}
                                <span>
                                  {shift.slotDuration} min appointments
                                </span>
                              </div>
                            </div>
                          ))}

                          <button
                            type="button"
                            className="add-shift-btn"
                            onClick={() => addShift(day)}
                          >
                            <PlusCircle size={16} />
                            Add another shift
                          </button>
                        </div>
                      )}
                    </section>
                  );
                })}
              </div>

              <div className="doctor-form-footer schedule-footer">
                <button
                  type="button"
                  className="doctor-cancel-btn"
                  onClick={() => setShowSchedule(false)}
                >
                  Cancel
                </button>

                <button type="submit" className="doctor-primary-btn">
                  Save Weekly Schedule
                </button>
              </div>
            </form>
          </div>
        </div>
      )}
    </div>
  );
}

import { useEffect, useMemo, useState } from "react";
import {
  Search,
  Plus,
  Users,
  Phone,
  MapPin,
  CalendarDays,
  Pencil,
  Trash2,
  X,
  UserRound,
} from "lucide-react";

import { demoStore } from "../../data/demoStore";
import "./Patients.css";

const emptyPatient = {
  name: "",
  dob: "",
  gender: "",
  phone: "",
  address: "",
  bloodGroup: "",
};

export default function Patients() {
  const [patients, setPatients] = useState([]);
  const [search, setSearch] = useState("");
  const [showModal, setShowModal] = useState(false);
  const [editingId, setEditingId] = useState(null);
  const [form, setForm] = useState(emptyPatient);

  const loadPatients = () => {
    setPatients(demoStore.getAll("patients"));
  };

  useEffect(() => {
    loadPatients();
    return demoStore.subscribe(loadPatients);
  }, []);

  const filteredPatients = useMemo(() => {
    const query = search.toLowerCase().trim();

    if (!query) return patients;

    return patients.filter((patient) =>
      [patient.id, patient.name, patient.phone, patient.address].some((value) =>
        String(value || "")
          .toLowerCase()
          .includes(query),
      ),
    );
  }, [patients, search]);

  const openAddModal = () => {
    setEditingId(null);
    setForm(emptyPatient);
    setShowModal(true);
  };

  const openEditModal = (patient) => {
    setEditingId(patient.id);
    setForm({
      name: patient.name || "",
      dob: patient.dob || "",
      gender: patient.gender || "",
      phone: patient.phone || "",
      address: patient.address || "",
      bloodGroup: patient.bloodGroup || "",
    });
    setShowModal(true);
  };

  const handleChange = (e) => {
    const { name, value } = e.target;
    setForm((prev) => ({ ...prev, [name]: value }));
  };

  const handleSubmit = (e) => {
    e.preventDefault();

    if (!form.name.trim() || !form.phone.trim()) {
      alert("Name and phone number are required.");
      return;
    }

    if (!/^\d{10}$/.test(form.phone)) {
      alert("Enter a valid 10-digit phone number.");
      return;
    }

    const payload = {
      ...form,
      name: form.name.trim(),
      phone: form.phone.trim(),
      address: form.address.trim(),
      registeredAt: editingId
        ? demoStore.getById("patients", editingId)?.registeredAt
        : new Date().toISOString().split("T")[0],
    };

    const result = editingId
      ? demoStore.update("patients", editingId, payload)
      : demoStore.add("patients", payload, "PAT");

    if (!result) {
      alert("Unable to save patient. Please try again.");
      return;
    }

    setShowModal(false);
    setForm(emptyPatient);
    setEditingId(null);
    loadPatients();
  };

  const handleDelete = (patient) => {
    const confirmed = window.confirm(
      `Are you sure you want to delete ${patient.name}?`,
    );

    if (!confirmed) return;

    demoStore.delete("patients", patient.id);
    loadPatients();
  };

  const calculateAge = (dob) => {
    if (!dob) return "—";

    const birthDate = new Date(dob);
    const today = new Date();

    let age = today.getFullYear() - birthDate.getFullYear();

    const monthDifference = today.getMonth() - birthDate.getMonth();

    if (
      monthDifference < 0 ||
      (monthDifference === 0 && today.getDate() < birthDate.getDate())
    ) {
      age--;
    }

    return age >= 0 ? `${age} yrs` : "—";
  };

  return (
    <div className="patients-page">
      <div className="patients-header">
        <div>
          <span className="patients-eyebrow">MANAGEMENT</span>
          <h1>Patients</h1>
          <p>Manage patient information and records.</p>
        </div>

        <button className="patient-add-btn" onClick={openAddModal}>
          <Plus size={18} />
          Add Patient
        </button>
      </div>

      <div className="patients-stats">
        <div className="patient-stat-card">
          <div className="patient-stat-icon blue">
            <Users size={21} />
          </div>
          <div>
            <span>Total Patients</span>
            <strong>{patients.length}</strong>
          </div>
        </div>

        <div className="patient-stat-card">
          <div className="patient-stat-icon green">
            <UserRound size={21} />
          </div>
          <div>
            <span>New This Month</span>
            <strong>
              {
                patients.filter((patient) => {
                  if (!patient.registeredAt) return false;

                  const date = new Date(patient.registeredAt);
                  const now = new Date();

                  return (
                    date.getMonth() === now.getMonth() &&
                    date.getFullYear() === now.getFullYear()
                  );
                }).length
              }
            </strong>
          </div>
        </div>

        <div className="patient-stat-card">
          <div className="patient-stat-icon purple">
            <CalendarDays size={21} />
          </div>
          <div>
            <span>Returning Patients</span>
            <strong>
              {(() => {
                const appointments = demoStore.getAll("appointments");

                const visitedPatientIds = new Set(
                  appointments
                    .filter((appointment) => appointment.status === "Completed")
                    .map((appointment) => appointment.patientId),
                );

                return visitedPatientIds.size;
              })()}
            </strong>
          </div>
        </div>
      </div>

      <div className="patients-content">
        <div className="patients-toolbar">
          <div>
            <h2>All Patients</h2>
            <p>{filteredPatients.length} records found</p>
          </div>

          <div className="patient-search">
            <Search size={18} />
            <input
              type="text"
              placeholder="Search patients..."
              value={search}
              onChange={(e) => setSearch(e.target.value)}
            />
          </div>
        </div>

        <div className="patients-table-wrapper">
          <table className="patients-table">
            <thead>
              <tr>
                <th>Patient</th>
                <th>Patient ID</th>
                <th>Age / Gender</th>
                <th>Contact</th>
                <th>Blood Group</th>
                <th>Registered</th>
                <th>Actions</th>
              </tr>
            </thead>

            <tbody>
              {filteredPatients.map((patient) => (
                <tr key={patient.id}>
                  <td>
                    <div className="patient-name-cell">
                      <div className="patient-avatar">
                        {patient.name?.charAt(0)?.toUpperCase()}
                      </div>
                      <div>
                        <strong>{patient.name}</strong>
                        <span>{patient.address || "No address"}</span>
                      </div>
                    </div>
                  </td>

                  <td>
                    <span className="patient-id">{patient.id}</span>
                  </td>

                  <td>
                    <div className="patient-age-gender">
                      <strong>{calculateAge(patient.dob)}</strong>
                      <span>{patient.gender || "—"}</span>
                    </div>
                  </td>

                  <td>
                    <div className="patient-contact">
                      <Phone size={14} />
                      {patient.phone}
                    </div>
                  </td>

                  <td>
                    <span className="blood-group">
                      {patient.bloodGroup || "—"}
                    </span>
                  </td>

                  <td>{patient.registeredAt || "—"}</td>

                  <td>
                    <div className="patient-actions">
                      <button
                        title="Edit patient"
                        onClick={() => openEditModal(patient)}
                      >
                        <Pencil size={16} />
                      </button>

                      <button
                        className="delete"
                        title="Delete patient"
                        onClick={() => handleDelete(patient)}
                      >
                        <Trash2 size={16} />
                      </button>
                    </div>
                  </td>
                </tr>
              ))}

              {filteredPatients.length === 0 && (
                <tr>
                  <td colSpan="7">
                    <div className="patients-empty">
                      <Users size={32} />
                      <strong>No patients found</strong>
                      <span>Try another search or add a patient.</span>
                    </div>
                  </td>
                </tr>
              )}
            </tbody>
          </table>
        </div>
      </div>

      {showModal && (
        <div
          className="patient-modal-overlay"
          onMouseDown={(e) => {
            if (e.target === e.currentTarget) {
              setShowModal(false);
            }
          }}
        >
          <div className="patient-modal">
            <div className="patient-modal-header">
              <div>
                <span className="patients-eyebrow">PATIENT MANAGEMENT</span>
                <h2>{editingId ? "Edit Patient" : "Register Patient"}</h2>
              </div>

              <button
                className="patient-modal-close"
                onClick={() => setShowModal(false)}
              >
                <X size={20} />
              </button>
            </div>

            <form onSubmit={handleSubmit}>
              <div className="patient-form-grid">
                <label>
                  Full Name *
                  <input
                    name="name"
                    value={form.name}
                    onChange={handleChange}
                    placeholder="Enter full name"
                    required
                  />
                </label>

                <label>
                  Date of Birth
                  <input
                    type="date"
                    name="dob"
                    value={form.dob}
                    max={new Date().toISOString().split("T")[0]}
                    onChange={handleChange}
                  />
                </label>

                <label>
                  Gender
                  <select
                    name="gender"
                    value={form.gender}
                    onChange={handleChange}
                  >
                    <option value="">Select gender</option>
                    <option value="Male">Male</option>
                    <option value="Female">Female</option>
                    <option value="Other">Other</option>
                  </select>
                </label>

                <label>
                  Phone Number *
                  <input
                    name="phone"
                    type="tel"
                    inputMode="numeric"
                    maxLength="10"
                    value={form.phone}
                    onChange={handleChange}
                    placeholder="10-digit mobile number"
                    required
                  />
                </label>

                <label>
                  Blood Group
                  <select
                    name="bloodGroup"
                    value={form.bloodGroup}
                    onChange={handleChange}
                  >
                    <option value="">Select blood group</option>
                    {["A+", "A-", "B+", "B-", "O+", "O-", "AB+", "AB-"].map(
                      (group) => (
                        <option key={group} value={group}>
                          {group}
                        </option>
                      ),
                    )}
                  </select>
                </label>

                <label className="patient-full-width">
                  Address
                  <textarea
                    name="address"
                    value={form.address}
                    onChange={handleChange}
                    placeholder="Enter residential address"
                    rows="3"
                  />
                </label>
              </div>

              <div className="patient-form-footer">
                <button
                  type="button"
                  className="patient-cancel-btn"
                  onClick={() => setShowModal(false)}
                >
                  Cancel
                </button>

                <button type="submit" className="patient-save-btn">
                  {editingId ? "Save Changes" : "Register Patient"}
                </button>
              </div>
            </form>
          </div>
        </div>
      )}
    </div>
  );
}

import {
    demoDoctors,
    demoPatients,
    demoAppointments,
    demoStaff,
    demoSchedules,
} from "./demoData";

const STORAGE_KEY = "clinic_management_demo_v1";

const initialData = {
    doctors: demoDoctors,
    patients: demoPatients,
    appointments: demoAppointments,
    staff: demoStaff,
    schedules: demoSchedules,
};

function clone(data) {
    return JSON.parse(JSON.stringify(data));
}

function readData() {
    try {
        const stored = localStorage.getItem(STORAGE_KEY);

        if (stored) {
            return JSON.parse(stored);
        }
    } catch (error) {
        console.error("Unable to read demo data:", error);
    }

    return null;
}

function writeData(data) {
    try {
        localStorage.setItem(STORAGE_KEY, JSON.stringify(data));
        window.dispatchEvent(new Event("clinic-data-updated"));
        return true;
    } catch (error) {
        console.error("Unable to save demo data:", error);
        return false;
    }
}

function initialize() {
    if (typeof window === "undefined") return;

    if (!localStorage.getItem(STORAGE_KEY)) {
        writeData(clone(initialData));
    }
}

initialize();

function getData() {
    return readData() || clone(initialData);
}

function getCollection(collection) {
    return getData()[collection] || [];
}

function saveCollection(collection, items) {
    const data = getData();
    data[collection] = items;
    return writeData(data);
}

function generateId(prefix, items) {
    const numbers = items
        .map((item) => Number(item.id?.replace(prefix, "")))
        .filter(Number.isFinite);

    const next = Math.max(0, ...numbers) + 1;

    return `${prefix}${String(next).padStart(3, "0")}`;
}

export const demoStore = {
    // Get all records
    getAll(collection) {
        return getCollection(collection);
    },

    // Find a record
    getById(collection, id) {
        return getCollection(collection).find(
            (item) => item.id === id
        ) || null;
    },

    // Add a record
    add(collection, record, prefix) {
        const items = getCollection(collection);

        const newRecord = {
            ...record,
            id: record.id || generateId(prefix, items),
        };

        const success = saveCollection(collection, [
            ...items,
            newRecord,
        ]);

        return success ? newRecord : null;
    },

    // Update a record
    update(collection, id, updates) {
        const items = getCollection(collection);

        const index = items.findIndex(
            (item) => item.id === id
        );

        if (index === -1) return null;

        const updatedRecord = {
            ...items[index],
            ...updates,
            id: items[index].id,
        };

        const updatedItems = [...items];
        updatedItems[index] = updatedRecord;

        return saveCollection(collection, updatedItems)
            ? updatedRecord
            : null;
    },

    // Delete a record
    delete(collection, id) {
        const items = getCollection(collection);

        const updatedItems = items.filter(
            (item) => item.id !== id
        );

        if (updatedItems.length === items.length) {
            return false;
        }

        return saveCollection(collection, updatedItems);
    },

    // Search records
    search(collection, query, fields = ["name"]) {
        const items = getCollection(collection);
        const normalizedQuery = query.toLowerCase().trim();

        if (!normalizedQuery) return items;

        return items.filter((item) =>
            fields.some((field) =>
                String(item[field] ?? "")
                    .toLowerCase()
                    .includes(normalizedQuery)
            )
        );
    },

    // Reset demo data
    reset() {
        return writeData(clone(initialData));
    },

    // Listen for changes
    subscribe(callback) {
        const handler = () => callback(getData());

        window.addEventListener("clinic-data-updated", handler);

        return () => {
            window.removeEventListener(
                "clinic-data-updated",
                handler
            );
        };
    },
};
